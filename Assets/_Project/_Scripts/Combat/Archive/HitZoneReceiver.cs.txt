// Unity 컴파일에서 제외한 이전 Combat 구현
using UnityEngine;

namespace UJam.Runtime.Combat
{
    public sealed class HitZoneReceiver : MonoBehaviour, IDamageable
    {
        // 부위 피해를 전달할 본체 DamageReceiver
        [SerializeField] private DamageReceiver _damageReceiver;

        // 본체에 전달할 피해 배율
        [SerializeField, Min(0.0001f)] private float _damageMultiplier = 1f;

        // 기존 HitZone 값 타입으로 만들 부위 식별자
        [SerializeField] private string _zoneId = "Default";

        // 현재 Component가 나타내는 HitZone 값
        public HitZone Zone
        {
            get
            {
                // Inspector 식별자로 HitZone 생성
                return new HitZone(_zoneId);
            }
        }

        // 부위 피해를 본체 DamageReceiver로 전달
        public DamageResult TakeDamage(DamageInfo info)
        {
            // 현재 부위 식별자 생성
            HitZone zone = Zone;

            // 본체 참조가 없으면 부위 피해를 차단
            if (_damageReceiver == null)
            {
                // 누락된 본체 연결을 차단 결과로 반환
                return new DamageResult(0f, true, false, zone);
            }

            // Inspector 피격 부위 식별자가 비어 있는지 확인
            if (zone.IsEmpty)
            {
                // 유효하지 않은 피격 부위를 차단 결과로 반환
                return new DamageResult(0f, true, false, zone);
            }

            // 부위 피해 배율이 양의 유한 값인지 확인
            if (!IsPositiveFinite(_damageMultiplier))
            {
                // 유효하지 않은 배율을 차단 결과로 반환
                return new DamageResult(0f, true, false, zone);
            }

            // 검증된 부위 피해 배율 저장
            float multiplier = _damageMultiplier;

            // 부위 배율을 적용한 RawDamage 계산
            float adjustedRawDamage = info.RawDamage * multiplier;

            // 기존 Source·Type·Element·Flags를 보존한 피격 문맥 생성
            HitContext adjustedContext = new HitContext(zone);

            // 부위 보정 정보를 새 DamageInfo로 구성
            DamageInfo adjustedInfo = new DamageInfo(
                info.Source,
                adjustedRawDamage,
                info.DamageType,
                info.Element,
                adjustedContext,
                info.Flags);

            // 보정된 피해를 본체에 전달한 결과 반환
            return _damageReceiver.TakeDamage(adjustedInfo);
        }

        // 피해 배율이 양의 유한 값인지 확인
        private static bool IsPositiveFinite(float value)
        {
            // NaN과 무한대 또는 0 이하 값인지 확인
            if (float.IsNaN(value) || float.IsInfinity(value) || value <= 0f)
            {
                // 유효하지 않은 값 결과 반환
                return false;
            }

            // 양의 유한 값 결과 반환
            return true;
        }
    }
}
