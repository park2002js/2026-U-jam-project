// Unity 컴파일에서 제외한 이전 Combat 구현
using System;

namespace UJam.Runtime.Combat
{
    public readonly struct AttackId
    {
        public AttackId(string value)
        {
            Value = value;
        }

        public string Value { get; }

        public bool IsEmpty => string.IsNullOrEmpty(Value);
    }
}
